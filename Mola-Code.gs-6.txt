function today_() { return Utilities.formatDate(new Date(), 'Europe/Istanbul', 'yyyyMMdd'); }
function all_() {
  const o = P.getProperties(), t = today_(), r = [];
  for (const k in o) if (k.indexOf('P_') === 0) {
    const s = JSON.parse(o[k]);
    if (s.day !== t) { s.day = t; s.done = []; s.active = null; }
    s.done = s.done || [];
    r.push(s);
  }
  return r;
}
function save_(s) { P.setProperty('P_' + s.id, JSON.stringify(s)); }
function find_(id) { const s = all_().find(x => x.id === id); if (!s) throw new Error('Personel bulunamadı'); return s; }
function auth_(pin) { if (String(pin) !== String(CFG.PIN)) throw new Error('Şifre yanlış'); }

function locked_(fn) { const l = LockService.getScriptLock(); l.waitLock(15000); try { return fn(); } finally { l.releaseLock(); } }
function total_(b) { return (CFG.SEQ[b] || CFG.SEQ.opt).reduce((a, q) => a + q.d, 0); }
function mins_(x) { return Number(x && typeof x === 'object' ? x.m : x) || 0; }
function mainT_(s) { return (s.done || []).filter(x => x && (x.k === 'mola' || x.k === 'yemek')).reduce((a, x) => a + mins_(x), 0); }
function usedK_(s, k) { return (s.done || []).filter(x => x && x.k === k).reduce((a, x) => a + mins_(x), 0); }
function md_() { const p = Utilities.formatDate(new Date(), 'Europe/Istanbul', 'HH:mm').split(':'); return +p[0] * 60 + +p[1]; }
// Şu an başlatılabilecek moladan: sıradaki ana mola (erken bitirilmişse KALAN süresiyle) + saat aralığındaki ek molalar
function opts_(s) {
  const r = [], T = mainT_(s), md = md_();
  let b = 0;
  if (md >= CFG.MAIN.from * 60 && md < CFG.MAIN.to * 60) for (const q of (CFG.SEQ[s.branch] || CFG.SEQ.opt)) { b += q.d; if (T < b) { r.push({ k: q.k, n: q.n, d: b - T, full: q.d }); break; } }
  CFG.EXTRA.forEach(e => { const d = e.d - usedK_(s, e.k); if (md >= e.from * 60 && md < e.to * 60 && d > 0) r.push({ k: e.k, n: e.n, d: d, full: e.d }); });
  return r;
}
function names_() { const o = {}; CFG.SEQ.opt.concat(CFG.EXTRA).forEach(q => o[q.k] = q.n); return o; }
function rnd_(s) { const T = mainT_(s); let b = 0, i = 0; for (const q of (CFG.SEQ[s.branch] || CFG.SEQ.opt)) { b += q.d; if (T < b) return i; i++; } return i; }
// ===== VARDİYA =====
function wd_() { return Number(Utilities.formatDate(new Date(), 'Europe/Istanbul', 'u')) - 1; }   // Pzt=0 ... Paz=6
function mm_(t) { const p = t.split(':'); return +p[0] * 60 + +p[1]; }
// Vardiya yüklenmemişse herkes görünür. Gece yarısını geçen vardiyalar (16:00-02:00) dahil.
function shift_(s, md, wd) {
  if (!s.sh) return { on: true, start: 0, txt: '' };
  const cell = c => { if (!c) return null; const p = c.split('-'); return { a: mm_(p[0]), b: mm_(p[1]), t: c }; };
  const t = cell(s.sh[wd]), y = cell(s.sh[(wd + 6) % 7]);
  if (t && (t.b > t.a ? (md >= t.a && md < t.b) : md >= t.a)) return { on: true, start: t.a, txt: t.t };
  if (y && y.b <= y.a && md < y.b) return { on: true, start: y.a - 1440, txt: y.t };
  return { on: false, start: t ? t.a : 9999, txt: t ? t.t : '' };
}
function norm_(s) {
  return String(s || '').replace(/İ/g, 'i').replace(/I/g, 'i').toLowerCase().replace(/ı/g, 'i').replace(/ğ/g, 'g').replace(/ü/g, 'u').replace(/ş/g, 's').replace(/ö/g, 'o').replace(/ç/g, 'c').replace(/[^a-z0-9]/g, '');
}
function parseCell_(c) {   // "08:00-16:00", "8-16", "08.00 - 16.00" -> "08:00-16:00"; izin/boş -> ''
  const m = String(c || '').match(/(\d{1,2})(?:[:.](\d{2}))?\s*[-–—]\s*(\d{1,2})(?:[:.](\d{2}))?/);
  if (!m) return '';
  const h1 = +m[1], i1 = +(m[2] || 0), h2 = +m[3], i2 = +(m[4] || 0);
  if (h1 > 23 || h2 > 24 || i1 > 59 || i2 > 59) return '';
  const f = (h, i) => (h < 10 ? '0' : '') + h + ':' + (i < 10 ? '0' : '') + i;
  return f(h1, i1) + '-' + f(h2, i2);
}
function end_(a) { return a.start + (a.pt || 0) + a.dur * 60000; }
function branchName_(b) { return b === 'mavi' ? 'Mavi Bahçe' : 'Optimum'; }
function topicOf_(id) { return CFG.NTFY_PREFIX + '-' + id; }
function copy_(o) { return JSON.parse(JSON.stringify(o)); }
function nice_(n) {
  return String(n || '').trim().split(/\s+/).map(w => {
    const l = w.replace(/I/g, 'ı').replace(/İ/g, 'i').toLowerCase();
    return l.charAt(0).replace('i', 'İ').replace('ı', 'I').toUpperCase() + l.slice(1);
  }).join(' ');
}
function phone_(n) {
  let s = String(n || '').replace(/[^\d+]/g, '');
  if (!s) return '';
  if (s[0] === '+') return s;
  if (s.indexOf('00') === 0) return '+' + s.slice(2);
  if (s[0] === '0') return '+90' + s.slice(1);
  if (s.length === 10 && s[0] === '5') return '+90' + s;
  return '+' + s;
}

// Garson ekranına e-posta/telefon gönderilmez
function getState(branch, pin) {
  const mgr = !!pin && checkPin(pin), md = md_(), wd = wd_();
  const staff = all_().filter(s => s.branch === branch).sort((a, b) => a.name.localeCompare(b.name, 'tr')).map(s => {
    const c = mgr ? s : copy_(s);
    const sh = shift_(s, md, wd); c.on = sh.on; c.shTxt = sh.txt; c.shStart = sh.start;
    c.opts = s.off ? [] : opts_(s);
    c.bitti = !s.off && !c.opts.length && (mainT_(s) >= total_(s.branch) || md_() >= CFG.MAIN.to * 60);
    if (!mgr) { delete c.email; delete c.phone; if (c.active) ['nid', 'err', 'mailErr'].forEach(k => delete c.active[k]); }
    return c;
  });
  // Mola sırası: vardiyaya ilk giren başta. Sıra yalnızca GÖSTERİLİR (zorunlu değil)
  const onl = staff.filter(c => c.on && !c.off).sort((a, b) => a.shStart - b.shStart || a.name.localeCompare(b.name, 'tr'));
  onl.forEach((c, i) => { if (c.sh) c.sira = i + 1; });
  if (staff.some(c => c.sh) && md >= CFG.MAIN.from * 60 && md < CFG.MAIN.to * 60) {
    const n = (CFG.SEQ[branch] || CFG.SEQ.opt).length;
    const cand = onl.filter(c => !c.active && rnd_(c) < n).sort((a, b) => rnd_(a) - rnd_(b) || (a.sira || 999) - (b.sira || 999));
    if (cand[0]) cand[0].turn = true;
  }
  return { now: Date.now(), names: names_(), total: total_(branch), staff: staff };
}

function token_() { return String(P.getProperty('NTFY_TOKEN') || CFG.NTFY_TOKEN || '').trim(); }
function authHeaders_() { const t = token_(); return t ? { Authorization: 'Bearer ' + t } : {}; }
function saveToken(token, pin) {
  auth_(pin);
  const t = String(token || '').trim();
  if (!t) throw new Error('Token boş');
  const r = UrlFetchApp.fetch('https://ntfy.sh/v1/account', { headers: { Authorization: 'Bearer ' + t }, muteHttpExceptions: true });
  const c = r.getResponseCode();
  if (c === 401 || c === 403) throw new Error('Token geçersiz. ntfy.sh/app > Account > Access tokens bölümünden tk_ ile başlayan token\'ı kopyalayın.');
  if (c !== 200) throw new Error('Token doğrulanamadı: ' + c);
  P.setProperty('NTFY_TOKEN', t);
  return 'ok';
}
function errText_(r) {
  const c = r.getResponseCode();
  if (c === 429) return token_() ? 'ntfy günlük mesaj kotası dolu (429)' : 'Token kayıtlı değil (429). Panelde ⚙ Araçlar > 🔑 Ntfy Token';
  if (c === 401 || c === 403) return 'Token geçersiz (' + c + ')';
  return c + ' ' + r.getContentText().replace(/\s+/g, ' ').slice(0, 70);
}
function post_(topic, title, text, delay) {   // {ok:true,id} ya da {ok:false,err}
  const body = { topic: topic, title: title, message: text, priority: 5, tags: ['rotating_light', 'alarm_clock'] };
  if (delay) body.delay = delay;   // "20m" ya da unix saniye: ntfy sunucusu o zaman teslim eder
  try {
    const r = UrlFetchApp.fetch('https://ntfy.sh/', { method: 'post', contentType: 'application/json', muteHttpExceptions: true,
      headers: authHeaders_(), payload: JSON.stringify(body) });
    if (r.getResponseCode() !== 200) return { ok: false, err: errText_(r) };
    let id = ''; try { id = JSON.parse(r.getContentText()).id || ''; } catch (e) {}
    return { ok: true, id: id };
  } catch (e) { return { ok: false, err: String(e).slice(0, 70) }; }
}
function cancel_(topic, nid) {
  if (!nid) return;
  try { UrlFetchApp.fetch('https://ntfy.sh/' + topic + '/' + nid, { method: 'delete', headers: authHeaders_(), muteHttpExceptions: true }); } catch (e) {}
}

// Bitiş bildirimi ntfy'ye ZAMANLI olarak YALNIZCA BİR KEZ bırakılır; başarılıysa tekrar gönderilmez
function schedule_(id, start, text, delay) {
  const m = post_(topicOf_('yonetici'), '⏰ Mola doldu', text, delay);
  locked_(() => {
    const c = all_().find(x => x.id === id);
    if (!c || !c.active || c.active.start !== start) { if (m.ok) cancel_(topicOf_('yonetici'), m.id); return; }
    c.active.posting = false;
    if (m.ok) { c.active.sched = true; c.active.nid = m.id; c.active.err = ''; } else { c.active.err = m.err; }
    save_(c);
  });
}
function msgOf_(s, a) { return `${nice_(s.name)} (${branchName_(s.branch)}) - ${a.dur} dk ${a.n} süresi doldu!`; }

function startBreak(id, k) {
  let s;
  locked_(() => {
    s = find_(id);
    if (s.off) throw new Error('Personel izinli');
    if (s.sh && !shift_(s, md_(), wd_()).on) throw new Error('Vardiyan dışındasın');
    if (s.active) throw new Error('Zaten molada');
    const o = opts_(s).find(x => x.k === k);
    if (!o) throw new Error('Bu mola şu an kullanılamaz');
    s.active = { k: o.k, n: o.n, start: Date.now(), dur: o.d, pt: 0, notified: false, sched: false, posting: true };
    save_(s);
  });
  schedule_(s.id, s.active.start, msgOf_(s, s.active), s.active.dur + 'm');
}
function pauseBreak(id, pin) {
  auth_(pin);
  let nid = '';
  locked_(() => {
    const s = find_(id), a = s.active;
    if (!a || a.paused) return;
    if (Date.now() >= end_(a)) throw new Error('Süre doldu. Molayı Bitir butonunu kullanın.');
    nid = a.nid;
    a.paused = true; a.pausedAt = Date.now(); a.sched = false; a.posting = false; a.nid = ''; a.sent = 0;
    save_(s);
  });
  cancel_(topicOf_('yonetici'), nid);   // zamanlı bildirimi iptal et
}
function resumeBreak(id, pin) {
  auth_(pin);
  let s = null;
  locked_(() => {
    const x = find_(id), a = x.active;
    if (!a || !a.paused) return;
    a.pt = (a.pt || 0) + Date.now() - a.pausedAt; a.paused = false; a.pausedAt = 0;
    a.notified = false; a.tries = 0; a.lastTry = 0; a.mailSent = false; a.err = '';
    a.posting = end_(a) - Date.now() > 15000;
    save_(x); s = x;
  });
  if (s && s.active.posting) schedule_(s.id, s.active.start, msgOf_(s, s.active), String(Math.round(end_(s.active) / 1000)));
}
function endBreak(id, pin) {   // gerçekte geçen süre günlük havuzdan düşer (fazla mola dahil)
  auth_(pin);
  let old = null;
  locked_(() => {
    const s = find_(id), a = s.active;
    if (!a) return;
    old = copy_(s);
    const e = a.paused ? a.pausedAt : Date.now(), m = Math.max(1, Math.round((e - a.start - (a.pt || 0)) / 60000));
    s.done.push({ k: a.k, m: m, t: a.start, e: e });
    s.active = null; save_(s);
    const h = JSON.parse(P.getProperty('H_' + s.id) || '[]');   // geçmiş: [başlangıç, tür, dakika]
    h.push([a.start, a.k, m, e]);
    P.setProperty('H_' + s.id, JSON.stringify(h.slice(-200)));
  });
  if (old) cancelActive_(old);
}

function uploadShift(branch, text, pin) {   // Excel/Sheets'ten yapıştırılan haftalık vardiya: isim, Pzt..Paz
  auth_(pin);
  const staff = all_().filter(s => s.branch === branch), got = {}, missing = [];
  String(text || '').split(/\r?\n/).forEach(line => {
    if (!line.trim()) return;
    let parts = line.split(/\t|;|\|/);
    if (parts.length < 2) parts = line.split(',');
    const name = String(parts[0] || '').trim();
    if (!name) return;
    const week = []; for (let i = 0; i < 7; i++) week.push(parseCell_(parts[i + 1]));
    const s = staff.find(x => norm_(x.name) === norm_(name));
    if (s) got[s.id] = week; else if (week.some(Boolean)) missing.push(name);
  });
  const ids = Object.keys(got);
  if (!ids.length) throw new Error('Hiç isim eşleşmedi. Her satır: personel adı, sonra Pazartesi-Pazar vardiyaları olmalı.');
  const absent = [];
  locked_(() => staff.forEach(x => {
    const s = find_(x.id);
    if (got[s.id]) s.sh = got[s.id]; else { s.sh = ['', '', '', '', '', '', '']; absent.push(s.name); }
    save_(s);
  }));
  return { ok: ids.length, missing: missing, absent: absent };
}

}

function trNum_(s) {
  s = String(s || '').replace(/\s/g, '');
  if (!s) return NaN;
  return s.indexOf(',') >= 0 ? Number(s.replace(/\./g, '').replace(',', '.')) : (/^\d{1,3}(\.\d{3})+$/.test(s) ? Number(s.replace(/\./g, '')) : Number(s));
}
function ocr_(blob) {   // Google Drive OCR: görseli geçici Google Dokümanı'na çevirir, metni alır, siler
  const h = { Authorization: 'Bearer ' + ScriptApp.getOAuthToken() }, b = 'mola' + Date.now();
  const pre = '--' + b + '\r\nContent-Type: application/json; charset=UTF-8\r\n\r\n' + JSON.stringify({ name: 'ocr-tmp', mimeType: 'application/vnd.google-apps.document' }) +
    '\r\n--' + b + '\r\nContent-Type: ' + blob.getContentType() + '\r\n\r\n';
  const body = Utilities.newBlob(pre).getBytes().concat(blob.getBytes()).concat(Utilities.newBlob('\r\n--' + b + '--').getBytes());
  const r = UrlFetchApp.fetch('https://www.googleapis.com/upload/drive/v3/files?uploadType=multipart&ocrLanguage=tr',
    { method: 'post', contentType: 'multipart/related; boundary=' + b, payload: body, headers: h, muteHttpExceptions: true });
  if (r.getResponseCode() >= 300) throw new Error(r.getResponseCode() === 403 ? 'Drive izni yok. Editörde izinVer fonksiyonunu bir kez çalıştırıp izin verin, sonra yeni sürüm dağıtın.' : 'OCR hatası ' + r.getResponseCode() + ': ' + r.getContentText().slice(0, 120));
  const id = JSON.parse(r.getContentText()).id;
  try {
    const x = UrlFetchApp.fetch('https://www.googleapis.com/drive/v3/files/' + id + '/export?mimeType=text%2Fplain', { headers: h, muteHttpExceptions: true });
    if (x.getResponseCode() >= 300) throw new Error('Metin alınamadı: ' + x.getResponseCode());
    return x.getContentText();
  } finally { UrlFetchApp.fetch('https://www.googleapis.com/drive/v3/files/' + id, { method: 'delete', headers: h, muteHttpExceptions: true }); }
}
function parseReport_(t) {
  const g = re => { const m = t.match(re); return m ? m[1].trim() : ''; };
  const name = g(/Ad[ıiI]\s*:\s*[\w\-]*\s*[\/|]\s*([^\n\r]+)/i), firma = g(/Firma\s+Ad[ıiI]\s*:\s*([^\n\r]+)/i);
  const range = g(/Tarih\s+Aral[ıiI][ğg][ıiI]\s*:\s*([\d.]+\s*-\s*[\d.]+)/i), rep = trNum_(g(/Ortalama\s+Hesap\s+Tutar\s*:?\s*([\d.]+,\d{2})/i));
  let cnt = '', amt = '', m; const re = /[ÖO]demeler\s+Toplam[ıiI]\s*:?\s*([\d.]+)\s+([\d.]+,\d{2})/gi;
  while ((m = re.exec(t))) { cnt = m[1]; amt = m[2]; }   // SONUNCUSU (saat aralığı tablosu) = Toplam Satış ve Hesap Sayısı ile aynıdır
  let hesap = trNum_(g(/Hesap\s+Say[ıiI]s[ıiI]\s*:?\s*([\d.]+)/i)); if (!(hesap > 0)) hesap = trNum_(cnt);
  let ciro = trNum_(amt); if (!(ciro > 0)) ciro = trNum_(g(/Toplam\s+Sat[ıiI][şs]\s*_?\s*:?\s*([\d.]+,\d{2})/i));
  return { name: name, firma: firma, range: range, rep: rep > 0 ? rep : '', hesap: hesap > 0 ? hesap : '', ciro: ciro > 0 ? ciro : '' };
}
function readReport(b64, pin) {
  auth_(pin);
  const text = ocr_(Utilities.newBlob(Utilities.base64Decode(b64), 'image/jpeg', 'rapor.jpg')), r = parseReport_(text);
  if (!r.name || !r.ciro || !r.hesap) r.raw = text.replace(/\s+/g, ' ').slice(0, 350);
  return r;
}
function saveReport(d, pin) {
  auth_(pin);
  const name = String(d.name || '').trim(), ciro = Number(d.ciro), hesap = Number(d.hesap);
  if (!name || !(ciro > 0) || !(hesap > 0)) throw new Error('Ad, ciro ve hesap sayısı gerekli');
  const range = String(d.range || '').trim(), key = norm_(name) + '_' + norm_(range), ort = Math.round(ciro / hesap * 100) / 100;
  P.setProperty('R_' + key, JSON.stringify({ key: key, name: name, firma: String(d.firma || '').trim(), range: range, ciro: ciro, hesap: hesap, ort: ort, t: Date.now() }));
  return { ort: ort };
}
function listReports(pin) {
  auth_(pin);
  const o = P.getProperties(), r = [];
  for (const k in o) if (k.indexOf('R_') === 0) r.push(JSON.parse(o[k]));
  return r.sort((a, b) => b.ort - a.ort);
}
function delReport(key, pin) { auth_(pin); P.deleteProperty('R_' + key); }

// ===== SÜRESİ DOLAN MOLALAR (dakikalık tetikleyici + açık sayfa) =====
function due_(s, now) {
  const a = s.active;
  return !!(a && !a.paused && !a.notified && now >= end_(a) && !(a.lastTry && now - a.lastTry < 60000));
}
function checkBreaks() {
  const todo = [];
  locked_(() => {
    const now = Date.now();
    all_().forEach(s => {
      if (!due_(s, now)) return;
      s.active.lastTry = now; s.active.tries = (s.active.tries || 0) + 1; save_(s);
      todo.push({ id: s.id, name: s.name, branch: s.branch, email: s.email, a: copy_(s.active) });
    });
  });
  todo.forEach(t => {
    const a = t.a, nm = nice_(t.name), now = Date.now();
    const toStaff = `${nm}, ${a.dur} dk ${a.n} süren doldu. Lütfen işe dön!`;
    const toMgr = `${nm} (${branchName_(t.branch)}) - ${a.dur} dk ${a.n} süresi doldu!`;
    let ok = !!a.sched, err = a.err || '';
    // zamanlı bildirim bırakılmışsa (ya da hâlâ bırakılıyorsa) tekrar GÖNDERİLMEZ -> tek bildirim
    if (!a.sched && !(a.posting && now - end_(a) < 120000)) {
      const m = post_(topicOf_('yonetici'), '⏰ Mola doldu', toMgr);
      ok = m.ok; err = m.err || '';
    }
    let mailErr = a.mailErr || '';
    if (!a.mailSent) mailErr = mail_(t.email, '⏰ Molan doldu - işe dön!', toStaff) || mail_(CFG.MANAGER_EMAIL, 'Mola doldu: ' + nm, toMgr);
    locked_(() => {
      const s = all_().find(x => x.id === t.id);
      if (!s || !s.active || s.active.start !== a.start) return;
      const c = s.active;
      c.mailSent = true; c.mailErr = mailErr;
      if (ok) { c.sent = c.sched ? end_(c) : Date.now(); c.err = ''; c.notified = true; }
      else { c.err = err; if ((c.tries || 0) >= 5) c.notified = true; }
      save_(s);
    });
  });
}
function setupTrigger() {   // Bir kez çalıştırın (mail iznini de ister)
  ScriptApp.getProjectTriggers().forEach(t => ScriptApp.deleteTrigger(t));
  ScriptApp.newTrigger('checkBreaks').timeBased().everyMinutes(1).create();
}
